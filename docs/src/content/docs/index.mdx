import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

<CardGrid>
	<Card title="Several APIs, one token each" icon="puzzle">
		Password or client-credentials grants, Keycloak or any token endpoint. A 401 renews the token
		and makes the call again once; settings shared by an environment are inherited field by field.
	</Card>
	<Card title="Waits on the database" icon="seti:db">
		A poll step runs a PostgreSQL query until a row shows up, then hands its columns to the next
		steps. Read-only connections, parameters bound, never concatenated.
	</Card>
	<Card title="Steps that are not needed" icon="approve-check">
		`skipIf` and `onlyIf` run a query before a step and leave it out when the work is already done,
		without counting it as a failure.
	</Card>
	<Card title="For each value" icon="list-format">
		One step, or a whole sequence, for each value of a list: all the generations, then all the
		waits, then all the downloads; or one value at a time.
	</Card>
	<Card title="A terminal UI, and a CLI" icon="laptop">
		Pick an environment, edit a variable, run everything or one step, and read the request and the
		response side by side. `--run` does the same in scripts and CI.
	</Card>
	<Card title="One native executable" icon="rocket">
		Built with GraalVM for Windows, Linux and macOS: no JVM to install, and
		`chainrunner self-update` installs the next release.
	</Card>
</CardGrid>
